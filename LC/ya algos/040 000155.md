**Задача 7 из 9 (блок 6): стек с минимумом** (LeetCode 155)

Нужно реализовать класс `MinStack`: обычный стек целых чисел, который умеет возвращать минимальный элемент за O(1).

```cpp
class MinStack {
public:
    void push(int value);
    void pop();
    int top() const;
    int get_min() const;
};
```

- `push(value)` кладёт значение на вершину.
- `pop()` убирает значение с вершины.
- `top()` возвращает значение на вершине, не убирая его.
- `get_min()` возвращает минимальное значение среди всех, которые сейчас лежат в стеке.

**Ограничения**

- `pop`, `top` и `get_min` вызываются только на непустом стеке.
- До 1 000 000 операций, значения лежат во всём диапазоне `int`, включая `INT_MIN` и `INT_MAX`, часто повторяются.
- Каждая операция работает за O(1).

2. Как сделать то же одним стеком пар `(значение, текущий минимум)`? Чем это лучше и хуже предыдущего варианта по памяти?

6. Как изменится решение, если нужен не минимум, а медиана?

	class MinStack {
	public:
		void push([[maybe_unused]] int value) {
		}

		void pop() {
		}

		int top() const {
			return 0;
		}

		int get_min() const {
			return 0;
		}
	};

	// kind: 'p' push value, 'o' pop, 't' check top, 'm' check get_min
	// (pop, top and get_min are never applied to an empty stack)
	struct Operation {
		char kind;
		int value;
	};

	// runs the operations on MinStack and on a model; the whole run must take less than two seconds
	void test_min_stack(const std::string& label, const std::vector<Operation>& operations) {
		MinStack stack;
		std::vector<int> values;
		std::vector<int> minimums;

		const auto start{std::chrono::steady_clock::now()};
		bool ok{true};
		for (std::size_t i{0}; ok && i < operations.size(); ++i) {
			const Operation& operation{operations[i]};
			switch (operation.kind) {
			case 'p':
				stack.push(operation.value);
				values.push_back(operation.value);
				minimums.push_back(minimums.empty() ? operation.value : std::min(minimums.back(), operation.value));
				break;
			case 'o':
				stack.pop();
				values.pop_back();
				minimums.pop_back();
				break;
			case 't':
				ok = stack.top() == values.back();
				break;
			default:
				ok = stack.get_min() == minimums.back();
				break;
			}
			if (i % 1024 == 0 && std::chrono::steady_clock::now() - start > std::chrono::seconds(2)) {
				ok = false;
			}
		}

		std::cout << std::format("{}: {}\n",
			label,
			ok ? "OK" : "FAIL") << std::flush;
	}

	// size values: pattern 0 grows, 1 decreases, 2 is constant, 3 is pseudo-random in the whole int range,
	// 4 alternates the minimum and the maximum int
	std::vector<int> make_values(int size, int pattern) {
		std::vector<int> values;
		values.reserve(static_cast<std::size_t>(size));
		unsigned state{12345u};
		for (int i{0}; i < size; ++i) {
			state = state * 1103515245u + 12345u;
			if (pattern == 0) values.push_back(i);
			else if (pattern == 1) values.push_back(size - i);
			else if (pattern == 2) values.push_back(7);
			else if (pattern == 3) values.push_back(static_cast<int>(state));
			else values.push_back(i % 2 == 0 ? 2147483647 : -2147483647 - 1);
		}
		return values;
	}

	// push all the values (asking for the minimum after every push), then pop them all
	// (asking for the top before every pop and for the minimum after it)
	std::vector<Operation> push_all_then_pop_all(const std::vector<int>& values) {
		std::vector<Operation> operations;
		operations.reserve(values.size() * 5);
		for (const int value : values) {
			operations.push_back({'p', value});
			operations.push_back({'m', 0});
		}
		for (std::size_t left{values.size()}; left > 0; --left) {
			operations.push_back({'t', 0});
			operations.push_back({'o', 0});
			if (left > 1) operations.push_back({'m', 0});
		}
		return operations;
	}

	// count pseudo-random operations; the stack never gets empty-popped, values are in [-range / 2, range / 2)
	std::vector<Operation> random_operations(int count, unsigned seed, int range) {
		std::vector<Operation> operations;
		operations.reserve(static_cast<std::size_t>(count) * 2);
		unsigned state{seed};
		std::size_t size{0};

			const unsigned choice{bits % 8};

				--size;
			} else {
				operations.push_back({'t', 0});
				operations.push_back({'m', 0});
			}
		}
		return operations;
	}
}

		std::stack<int> st{};

			}

			const auto top{st.top()};

			}

		}
