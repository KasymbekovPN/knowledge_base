
Блок 2, повторение 4 из 6: **Merge Intervals** (LC 56). Пишем с нуля, без подсказок.

**Условие.** Дан набор отрезков `intervals`, у каждого `[start, end]`, где `start <= end`. Объедините все пересекающиеся отрезки и верните список непересекающихся отрезков, покрывающих те же точки. Отрезки, касающиеся в одной точке (`[1, 4]` и `[4, 5]`), считаются пересекающимися и объединяются. Отрезки `[1, 2]` и `[3, 4]` не пересекаются и остаются раздельными. Результат нужно вернуть отсортированным по `start`.

**Сигнатура** (уже стоит в шаблоне как заглушка):

```cpp
std::vector<std::vector<int>> merge_intervals(std::vector<std::vector<int>>& intervals);
```

**Ограничения:** `1 <= intervals.size() <= 200000`, `0 <= start <= end <= 1e9`. Входной массив менять можно.

**Вопросы интервьюера:**

1. Какая сложность по времени и по памяти?
2. Почему после сортировки достаточно сравнить текущий отрезок только с последним в ответе?
3. Почему при слиянии нужно брать `max` концов, а не просто конец нового отрезка?
4. Что изменится, если отрезки приходят потоком и нужно держать объединённый набор актуальным после каждого добавления?

**Цель:** O(n log n) по времени, O(n) на ответ (сортировка на месте). O(n²) на больших тестах не уложится по времени. Если программа зависла, прервите её вручную, такой тест не прерывается сам.

	std::vector<std::vector<int>> expected_merge(
			const std::vector<std::vector<int>>& intervals) {
		// events: kind 0 = start, kind 1 = end; a start goes before an end at the same point
		std::vector<std::pair<int, int>> events;
		events.reserve(intervals.size() * 2);
		for (const auto& interval : intervals) {
			events.emplace_back(interval[0], 0);
			events.emplace_back(interval[1], 1);
		}
		std::ranges::sort(events);

		std::vector<std::vector<int>> result;
		int depth{0};
		int begin{0};
		for (const auto& [point, kind] : events) {
			if (kind == 0) {
				if (depth == 0) begin = point;
				++depth;
			} else {
				--depth;
				if (depth == 0) result.push_back({begin, point});
			}
		}
		return result;
	}

	void shuffle_intervals(std::vector<std::vector<int>>& intervals, unsigned seed) {

	std::vector<std::vector<int>> merge_intervals(
			std::vector<std::vector<int>>& intervals) {

		std::ranges::sort(intervals);

		std::vector<std::vector<int>> answer;
		for (std::size_t i{0}; i < intervals.size(); ++i) {
			if (answer.empty() || answer.back()[1] < intervals[i][0]) {
				answer.push_back(std::move(intervals[i]));
			} else {
				answer.back()[1] = std::max(answer.back()[1], intervals[i][1]);
			}
		}

		return answer;
	}

			const int start{static_cast<int>((state >> 8) % 100000u)};
			state = state * 1103515245u + 12345u;
			const int length{static_cast<int>((state >> 8) % 3u)};
			intervals.push_back({start, start + length});
		}
		test_merge_intervals("n = 200000, random on a small range", intervals);
	}
	{
		// n = 200000, all intervals are [7, 7]
		test_merge_intervals("n = 200000, identical point intervals",
			std::vector<std::vector<int>>(200000, std::vector<int>{7, 7}));
	}
	{
		// n = 200000, one huge interval hidden in the middle of many small ones
		std::vector<std::vector<int>> intervals;
		for (int i{0}; i < 199999; ++i) intervals.push_back({10 * i, 10 * i + 5});
		intervals.insert(intervals.begin() + 100000, std::vector<int>{0, 1000000000});
		test_merge_intervals("n = 200000, one huge interval among small ones", intervals);
	}
