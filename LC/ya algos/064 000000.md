**Блок 8, задача 6: LFU Cache**

**Условие.** Реализуйте кэш фиксированной ёмкости, который при переполнении вытесняет наименее часто используемый ключ. Любое обращение к ключу (`get` или `put`) считается одним использованием.

- `get(key)` возвращает значение или `std::nullopt`, если ключа нет. Найденный ключ получает ещё одно использование.
- `put(key, value)` для существующего ключа обновляет значение и тоже считает использование.
- `put(key, value)` для нового ключа, если кэш полон, сначала вытесняет ключ с наименьшим числом использований. Если таких несколько, вытесняется тот, к которому обращались давнее всех. Новый ключ начинает с одного использования.
- Ёмкость 0 означает, что кэш ничего не хранит.

**Сигнатура:**

```cpp
class LfuCache {
public:
    explicit LfuCache(std::size_t capacity);
    std::optional<int> get(int key);
    void put(int key, int value);
};
```

**Ограничения:**

- ёмкость от 0 до 10^5, до 10^6 операций, ключи и значения — весь диапазон `int`;
- каждая операция — O(1) в среднем.

**Пример** (ёмкость 2): `put(1,1)`, `put(2,2)`, `get(1)` даёт `1`, `put(3,3)` вытесняет ключ 2 (у него одно использование против двух у ключа 1), `get(2)` даёт `nullopt`, `get(3)` даёт `3`, `put(4,4)` вытесняет ключ 1 (у ключей 1 и 3 по два использования, ключ 1 использовался давнее), `get(1)` даёт `nullopt`, `get(3)` даёт `3`, `get(4)` даёт `4`.

// Least Frequently Used cache. Every get and every put of a key counts as one use.
// When the cache is full, put of a NEW key first evicts the key with the lowest
// number of uses; among such keys the least recently used one goes first.
// A new key starts with one use. Capacity 0 means the cache stores nothing.
class LfuCache {
	using it_type = std::list<std::pair<int, int>>::iterator;
	using key_type = int;
	using value_type = int;

	std::size_t capacity_{0};
	std::list<std::pair<key_type, value_type>> order_{};
	std::unordered_map<key_type, it_type> index_{};

	void touch(it_type it) {
		order_.splice(order_.begin(), order_, it);
	}

public:
    explicit LfuCache(std::size_t capacity): capacity_{capacity} {}

    std::optional<value_type> get(key_type key) {
	    const auto it{index_.find(key)};
	    if (it == index_.end()) { return std::nullopt; }
    
		touch(it->second);
		return it->second->second;
	}

    void put(key_type key, value_type value) {
	    const auto it{index_.find(key)};
	    
	    if (it != index_.end()) {
		    it->second->second = value;
		    touch(it->second);
		    return;
	    }
	    
	    if (order_.size() >= capacity) {
		    auto& lru{order_.back()};
		    index_.erase(lru->first);
		    order_.pop_back();
	    }
	    
	    order_.emplace(key, value);
	    index_[key] order_.begin();
    }
};

class LfuCache {
	struct Entry {
		int key;
		int value;
		std::size_t uses;
	};
	using List = std::list<Entry>;

	std::size_t capacity_;
	std::size_t min_uses_{0};
	std::unordered_map<std::size_t, List> by_uses_{};   // частота -> ключи, front = самый давний
	std::unordered_map<int, List::iterator> index_{};   // ключ -> позиция в своём списке

	// переносит элемент в список с частотой на единицу больше
	void touch(const List::iterator it) {
		const std::size_t old_uses{it->uses};
		List& old_list{by_uses_[old_uses]};
		List& new_list{by_uses_[old_uses + 1]};
		new_list.splice(new_list.end(), old_list, it);   // итератор остаётся валидным
		++it->uses;
		if (old_list.empty()) {
			by_uses_.erase(old_uses);
			if (min_uses_ == old_uses) { min_uses_ = old_uses + 1; }
		}
	}

public:
	explicit LfuCache(std::size_t capacity) : capacity_{capacity} {}

	std::optional<int> get(int key) {
		const auto found{index_.find(key)};
		if (found == index_.end()) { return std::nullopt; }
		touch(found->second);
		return found->second->value;
	}

	void put(int key, int value) {
		if (capacity_ == 0) { return; }

		const auto found{index_.find(key)};
		if (found != index_.end()) {
			found->second->value = value;
			touch(found->second);
			return;
		}

		if (index_.size() == capacity_) {
			List& victims{by_uses_[min_uses_]};
			index_.erase(victims.front().key);
			victims.pop_front();
			if (victims.empty()) { by_uses_.erase(min_uses_); }
		}

		List& fresh{by_uses_[1]};
		fresh.push_back({key, value, 1});
		index_[key] = std::prev(fresh.end());
		min_uses_ = 1;
	}
};

// Independent reference: entries are ordered by (uses, time of last use).
class Model {
public:
    explicit Model(std::size_t capacity) : capacity_(capacity) {}

    std::optional<int> get(int key) {
        const auto it = entries_.find(key);
        if (it == entries_.end()) return std::nullopt;
        touch(it);
        return it->second.value;
    }

    void put(int key, int value) {
        if (capacity_ == 0) return;
        const auto it = entries_.find(key);
        if (it != entries_.end()) {
            it->second.value = value;
            touch(it);
            return;
        }
        if (entries_.size() == capacity_) {
            const auto victim = order_.begin();
            entries_.erase(std::get<2>(*victim));
            order_.erase(victim);
        }
        entries_[key] = {value, 1, ++clock_};
        order_.insert({1, clock_, key});
    }

private:
    struct Entry {
        int value;
        int uses;
        std::int64_t stamp;
    };

    void touch(std::map<int, Entry>::iterator it) {
        order_.erase({it->second.uses, it->second.stamp, it->first});
        ++it->second.uses;
        it->second.stamp = ++clock_;
        order_.insert({it->second.uses, it->second.stamp, it->first});
    }

    std::size_t capacity_;
    std::int64_t clock_ = 0;
    std::map<int, Entry> entries_;
    std::set<std::tuple<int, std::int64_t, int>> order_;
};

void test_script(const std::string& label, std::size_t capacity, const std::vector<Op>& ops) {
    LfuCache cache(capacity);
    Model model(capacity);
    bool ok = true;
    for (const Op& op : ops) {
        if (op.is_put) {
            cache.put(op.key, op.value);
            model.put(op.key, op.value);
        } else {
            ok = (cache.get(op.key) == model.get(op.key)) && ok;
        }
    }
    report(label, ok);
}

    const bool fast = elapsed < std::chrono::seconds(1);

}
