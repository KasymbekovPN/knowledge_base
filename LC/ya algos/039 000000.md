	bool expected_valid(const std::string& s) {
		std::vector<char> opened;
		for (const char c : s) {

				opened.push_back(c);

			}

namespace {
	std::unordered_map<char, char> close_by_open = {
		{'(', ')'},
		{'[', ']'},
		{'{', '}'}
	};

	bool is_valid(const std::string& s) {
		const std::size_t size{s.size()};
		if (size == 0) return true;
		if (size % 2 == 1) return false;
	
		const auto first = s.front();
		if (first == ')' || first == ']' || first == '}') return false;
		
		std::stack<char> st;
		for (const auto ch: s) {
			if (ch == '(' || ch == '{' || ch == '[') {
				st.push(close_by_open[ch]);
				continue;
			}

			if (st.size() == 0) return false;

			const auto waited_close{st.top()};
			st.pop();
			if (waited_close != ch) return false;
		}
	
		return st.empty();
	}
