**Блок 8, задача 8: Median from Data Stream**

**Условие.** Реализуйте структуру, которая принимает поток целых чисел и в любой момент отвечает, какова медиана всех принятых чисел.

- `add_num(num)` добавляет число в поток.
- `find_median()` возвращает медиану. При нечётном количестве чисел это средний элемент в отсортированном порядке, при чётном — среднее двух средних. Вызывается только после хотя бы одного `add_num`.

**Сигнатура:**

```cpp
class MedianFinder {
public:
    void add_num(int num);
    double find_median() const;
};
```

**Ограничения:**

- до 10^6 операций, значения — весь диапазон `int`, возможны повторы;
- вызовов `find_median` может быть столько же, сколько вставок;
- `add_num` — O(log n), `find_median` — O(1).

**Пример:** `add_num(1)`, `add_num(2)`, `find_median()` даёт `1.5`, `add_num(3)`, `find_median()` даёт `2.0`.

// Median of a stream of integers. find_median is called only after at least one add_num.
// For an odd count it is the middle element, for an even count the average of the two middle ones.
class MedianFinder {
	std::priority_queue<int, std::vector<int>> min_heap{};
	std::priority_queue<int, std::vector<int>, std::greater<>> max_heap{};

	int get_top_and_pop(std::priority_queue<int>& pq) {
		const int value{pq.top};
		pq.pop();

		return value;
	}

public:
    void add_num(int) {
	    max_heap.push(num);
	    
	    if (!min_heap.empty() && max_heap.top() > min_heap.top()) {
		    const int max_top{get_top_and_pop(max_heap)};
		    const int min_top{get_top_and_pop(min_heap)};
		    max_heap.push(min_top);
		    min_heap.push(max_top);
	    }
	    
	    if (min_heap.size() > max_heap.size() + 1) {
		    max_heap.push(get_top_and_pop(min_heap));
	    } else if (max_heap.size() > min_heap.size() + 1) {
		    min_heap.push(get_top_and_pop(max_heap));
	    }
    }

    double find_median() const {
	    const std::size_t max_heap_size{max_heap.size()};
	    const std::size_t min_heap_size{min_heap.size()};
	    if (max_heap_size == min_heap_size) {
		    return static_cast<double>(max_heap.top()) +
				   static_cast<double>(min_heap.top()) / 2.0;
	    } else if (max_heap_size > min_heap_size) {
		    return static_cast<double>(max_heap.top());
	    }
		return static_cast<double>(min_heap.top());
	}
};

// Independent reference: a Fenwick tree over the compressed values of the whole script,
// the k-th smallest element is found by descending the tree.
class Model {
public:
    explicit Model(const std::vector<Op>& ops) {
        for (const Op& op : ops) if (op.is_add) values_.push_back(op.value);
        std::sort(values_.begin(), values_.end());
        values_.erase(std::unique(values_.begin(), values_.end()), values_.end());
        tree_.assign(values_.size() + 1, 0);
        top_bit_ = 1;
        while (top_bit_ * 2 <= values_.size()) top_bit_ *= 2;
    }

    void add(int value) {
        const std::size_t pos = static_cast<std::size_t>(
            std::lower_bound(values_.begin(), values_.end(), value) - values_.begin()) + 1;
        for (std::size_t i = pos; i < tree_.size(); i += i & (~i + 1)) ++tree_[i];
        ++count_;
    }

    double median() const {
        if (count_ % 2 == 1) return static_cast<double>(kth((count_ + 1) / 2));
        const long long sum = static_cast<long long>(kth(count_ / 2)) + kth(count_ / 2 + 1);
        return static_cast<double>(sum) / 2.0;
    }

private:
    int kth(std::size_t k) const {
        std::size_t pos = 0;
        for (std::size_t bit = top_bit_; bit > 0; bit /= 2) {
            const std::size_t next = pos + bit;
            if (next < tree_.size() && tree_[next] < k) {
                pos = next;
                k -= tree_[next];
            }
        }
        return values_[pos];
    }

    std::vector<int> values_;
    std::vector<std::size_t> tree_;
    std::size_t top_bit_ = 1;
    std::size_t count_ = 0;
};

            model.add(op.value);

}

// `queries_every`: one find_median after every that many insertions.
std::vector<Op> stream(std::uint32_t seed, int count, int queries_every, int mode) {
    std::mt19937 gen(seed);
    std::vector<Op> ops;
    for (int i = 0; i < count; ++i) {
        int value = 0;
        switch (mode) {
            case 0: value = static_cast<int>(gen()); break;                       // the whole int range
            case 1: value = i; break;                                             // ascending
            case 2: value = count - i; break;                                     // descending
            case 3: value = static_cast<int>(gen() % 5); break;                   // few distinct values
            default: value = (i % 2 == 0) ? INT_MAX - i / 2 : INT_MIN + i / 2; break;  // extremes
        }
        ops.push_back(A(value));
        if ((i + 1) % queries_every == 0) ops.push_back(M());
    }
    return ops;
}
