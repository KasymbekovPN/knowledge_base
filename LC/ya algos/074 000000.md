**Условие.** Реализуйте стек, в котором кроме `push`, `pop` и `top` есть получение минимума и максимума всех элементов стека за O(1).

- `push(value)` кладёт значение.
- `pop()` убирает верхний элемент.
- `top()` возвращает верхний элемент.
- `get_min()` и `get_max()` возвращают минимум и максимум среди элементов, находящихся в стеке сейчас.
- `pop`, `top`, `get_min` и `get_max` вызываются только на непустом стеке.

**Сигнатура:**

```cpp
class MinMaxStack {
public:

    int top() const;
    int get_min() const;
    int get_max() const;
```

```cpp
// pop, top, get_min and get_max are called only on a non-empty stack.
class MinMaxStack {
	struct Node {
		int val;
		int min;
		int max;
	};
	std::stack<int, std::vector<Node>> storage_{};

public:
    void push(const int value) {
	    if (storage_.empty()) {
		    storage_.push({value, value, value});
		    return;
	    }

	    storage_.push({
		    value,
			std::min(value, storage_.top().min),
			std::max(value, storage_.top().max)});
    }

    void pop() { storage_.pop(); }

    int top() const { return storage_.top().val; }

    int get_min() const { return storage_.top().min; }

    int get_max() const { return storage_.top().max; }
};

struct Op {
    char kind;  // 'P' push, 'O' pop, 'T' top, 'N' get_min, 'X' get_max
    int value;
};

Op P(int v) { return {'P', v}; }
Op O() { return {'O', 0}; }
Op T() { return {'T', 0}; }
Op N() { return {'N', 0}; }
Op X() { return {'X', 0}; }

void report(const std::string& label, bool ok) {
    std::cout << label << ": " << (ok ? "OK" : "FAIL") << '\n';
}

// Applies the operation; results of pop/push are 0.
int apply(MinMaxStack& s, const Op& op) {
    switch (op.kind) {
        case 'P': s.push(op.value); return 0;
        case 'O': s.pop(); return 0;
        case 'T': return s.top();
        case 'N': return s.get_min();
        default: return s.get_max();
    }
}

// Reference answers: a plain vector, minimum and maximum found by a linear scan.
std::vector<int> expected_answers(const std::vector<Op>& ops) {
    std::vector<int> data;
    std::vector<int> answers(ops.size(), 0);
    for (std::size_t i = 0; i < ops.size(); ++i) {
        switch (ops[i].kind) {
            case 'P': data.push_back(ops[i].value); break;
            case 'O': data.pop_back(); break;
            case 'T': answers[i] = data.back(); break;
            case 'N': answers[i] = *std::min_element(data.begin(), data.end()); break;
            default: answers[i] = *std::max_element(data.begin(), data.end()); break;
        }
    }
    return answers;
}

void test_script(const std::string& label, const std::vector<Op>& ops) {
    const std::vector<int> expected = expected_answers(ops);
    MinMaxStack s;
    bool ok = true;
    for (std::size_t i = 0; i < ops.size(); ++i) {
        const int got = apply(s, ops[i]);
        if (ops[i].kind != 'P' && ops[i].kind != 'O' && got != expected[i]) ok = false;
    }
    report(label, ok);
}

// For big tests the expected answers come from prefix minima/maxima kept next to the data.
void test_big(const std::string& label, const std::vector<Op>& ops) {
    std::vector<int> answers(ops.size(), 0);
    const auto start = std::chrono::steady_clock::now();
    {
        MinMaxStack s;
        for (std::size_t i = 0; i < ops.size(); ++i) answers[i] = apply(s, ops[i]);
    }
    const auto elapsed = std::chrono::steady_clock::now() - start;

    std::vector<int> data, mins, maxs;
    bool correct = true;
    for (std::size_t i = 0; i < ops.size() && correct; ++i) {
        switch (ops[i].kind) {
            case 'P':
                data.push_back(ops[i].value);
                mins.push_back(mins.empty() ? ops[i].value : std::min(mins.back(), ops[i].value));
                maxs.push_back(maxs.empty() ? ops[i].value : std::max(maxs.back(), ops[i].value));

    }
```
