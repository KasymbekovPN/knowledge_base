**Задача 5 блока 5 (бинарный поиск): Search a 2D Matrix**

**Условие.** Дана матрица `matrix` целых чисел размера `m × n` со свойствами:

- числа в каждой строке идут по неубыванию;
- первое число каждой строки строго больше последнего числа предыдущей строки.

Верните `true`, если `target` есть в матрице, иначе `false`.

**Сигнатура:**

```cpp
bool search_matrix(const std::vector<std::vector<int>>& matrix, int target);
```

**Ограничения:**

- m ≥ 1, n ≥ 1, m · n ≤ 1 000 000 (возможна одна строка из миллиона чисел или один столбец)
- `-2^31 ≤ matrix[i][j], target ≤ 2^31 - 1`
- все строки одной длины

**Примеры:**

- `matrix = [[1,3,5,7],[10,11,16,20],[23,30,34,60]], target = 3` → `true`
- `matrix = [[1,3,5,7],[10,11,16,20],[23,30,34,60]], target = 13` → `false`

**Вопросы интервьюера (ответьте после решения):**

1. Как вы превращаете одну позицию `mid` в пару (строка, столбец)? Что произойдёт, если перепутать делитель, и для каких форм матрицы (квадратная, одна строка, один столбец) ошибка не видна?
2. Можно ли искать, не выписывая матрицу в один массив? Какая сложность и какая дополнительная память у вашего подхода?
3. Почему не подходит поиск по строкам с последующим бинарным поиском внутри строки? Какова его сложность и на какой форме матрицы он станет слишком медленным?
4. Что изменится, если у второго условия убрать «строго» (первое число строки не меньше последнего числа предыдущей)? Что останется верным?

**Цель:** O(log(m · n)) по времени, O(1) по памяти.

namespace {
	bool search_matrix([[maybe_unused]] const std::vector<std::vector<int>>& matrix,
					   [[maybe_unused]] int target) {
		return false;
	}

	bool expected_found(const std::vector<std::vector<int>>& matrix, int target) {
		// the rows are sorted and every row starts above the end of the previous one,
		// so the matrix read row by row is one sorted array
		std::vector<int> flat;
		for (const auto& row : matrix) flat.insert(flat.end(), row.begin(), row.end());
		return std::ranges::binary_search(flat, target);
	}

	void test_matrix(const std::string& label,
					 const std::vector<std::vector<int>>& matrix,
					 int target) {
		const auto expected{expected_found(matrix, target)};

		const auto result{search_matrix(matrix, target)};

		std::cout << std::format("{}: {}\n",
			label,
			result == expected ? "OK" : "FAIL") << std::flush;
	}

	// many queries against one matrix: a scan of rows or of the whole matrix cannot pass this
	void test_matrix_many(const std::string& label,
						  const std::vector<std::vector<int>>& matrix,
						  const std::vector<int>& targets) {
		std::vector<int> flat;
		for (const auto& row : matrix) flat.insert(flat.end(), row.begin(), row.end());

		bool ok{true};
		for (const int target : targets) {
			if (search_matrix(matrix, target) != std::ranges::binary_search(flat, target)) {
				ok = false;
				break;
			}
		}

		std::cout << std::format("{}: {}\n",
			label,
			ok ? "OK" : "FAIL") << std::flush;
	}

	// rows x cols matrix filled row by row with values growing by `step`
	std::vector<std::vector<int>> make_matrix(int rows, int cols, int step) {
		std::vector<std::vector<int>> matrix(static_cast<std::size_t>(rows),
											 std::vector<int>(static_cast<std::size_t>(cols)));
		int value{-500000};
		for (auto& row : matrix) {
			for (auto& cell : row) {
				cell = value;
				value += step;
			}
		}
		return matrix;
	}

	std::vector<int> make_targets(int count, int low, int high, unsigned seed) {
		std::vector<int> targets;
		unsigned state{seed};
		for (int i{0}; i < count; ++i) {
			state = state * 1103515245u + 12345u;
			targets.push_back(low + static_cast<int>((state >> 4) % static_cast<unsigned>(high - low + 1)));
		}
		return targets;
	}
}

#include <iostream>
#include <format>
#include <vector>
#include <string>
#include <cstddef>

namespace {
	bool search_matrix(const std::vector<std::vector<int>>& matrix,
					   int target) {
		int rows{static_cast<int>(matrix.size())};
		int cols{static_cast<int>(matrix[0].size())};
		
		int lo{0};
		int hi{rows * cols - 1};
		
		while (lo <= hi) {
			int mid{lo + (hi - lo) / 2};
			const auto row_idx{static_cast<std::size_t>(mid / cols)};
			const auto col_idx{static_cast<std::size_t>(mid % cols)};
			const auto value{matrix[row_idx][col_idx]};
			
			if (value == target) {
				return true;
			} else if (value < target) {
				lo = mid + 1;
			} else {
				hi = mid - 1;
			}
		}
		
		return false;
	}

	bool expected_found(const std::vector<std::vector<int>>& matrix, int target) {
		// the rows are sorted and every row starts above the end of the previous one,
		// so the matrix read row by row is one sorted array
		std::vector<int> flat;
		for (const auto& row : matrix) flat.insert(flat.end(), row.begin(), row.end());
		return std::ranges::binary_search(flat, target);
	}

	void test_matrix(const std::string& label,
					 const std::vector<std::vector<int>>& matrix,
					 int target) {
		const auto expected{expected_found(matrix, target)};

		const auto result{search_matrix(matrix, target)};

		std::cout << std::format("{}: {}\n",
			label,
			result == expected ? "OK" : "FAIL") << std::flush;
	}

	// many queries against one matrix: a scan of rows or of the whole matrix cannot pass this
	void test_matrix_many(const std::string& label,
						  const std::vector<std::vector<int>>& matrix,
						  const std::vector<int>& targets) {
		std::vector<int> flat;
		for (const auto& row : matrix) flat.insert(flat.end(), row.begin(), row.end());

		bool ok{true};
		for (const int target : targets) {
			if (search_matrix(matrix, target) != std::ranges::binary_search(flat, target)) {
				ok = false;
				break;
			}
		}

		std::cout << std::format("{}: {}\n",
			label,
			ok ? "OK" : "FAIL") << std::flush;
	}

	// rows x cols matrix filled row by row with values growing by `step`
	std::vector<std::vector<int>> make_matrix(int rows, int cols, int step) {
		std::vector<std::vector<int>> matrix(static_cast<std::size_t>(rows),
											 std::vector<int>(static_cast<std::size_t>(cols)));
		int value{-500000};
		for (auto& row : matrix) {
			for (auto& cell : row) {
				cell = value;
				value += step;
			}
		}
		return matrix;
	}

	std::vector<int> make_targets(int count, int low, int high, unsigned seed) {
		std::vector<int> targets;

		for (int i{0}; i < count; ++i) {

			targets.push_back(low + static_cast<int>((state >> 4) % static_cast<unsigned>(high - low + 1)));
		}
		return targets;
	}
}
