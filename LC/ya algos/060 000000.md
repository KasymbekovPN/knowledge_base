**Блок 8, задача 2: Insert Delete GetRandom O(1)**

**Условие.** Реализуйте класс `RandomizedSet` — множество целых чисел с тремя операциями:

- `insert(val)` добавляет `val`. Возвращает `true`, если его не было, и `false`, если он уже лежал в множестве.
- `remove(val)` удаляет `val`. Возвращает `true`, если он был, и `false`, если его не было.
- `get_random(rng)` возвращает случайный элемент множества. Каждый элемент должен выпадать с одинаковой вероятностью. Генератор передаётся снаружи, чтобы тесты были воспроизводимы, и свой генератор создавать не нужно.

**Сигнатура:**

```cpp
class RandomizedSet {
public:
    bool insert(int val);
    bool remove(int val);
    int get_random(std::mt19937& rng);
};
```

**Ограничения:**

- до 10^6 операций, значения — весь диапазон `int`;
- `get_random` вызывается только на непустом множестве;
- каждая операция должна работать за O(1) в среднем.

3. Как изменится решение, если в множестве разрешены повторы (multiset)?

**Целевая сложность:** O(1) в среднем на операцию, O(n) памяти.

namespace {

class RandomizedSet {
public:
    bool insert(int) { return false; }
    bool remove(int) { return false; }
    int get_random(std::mt19937&) { return 0; }
};

struct Op {
    char kind;  // 'I' insert, 'R' remove, 'Q' get_random
    int value;
};

Op I(int v) { return {'I', v}; }
Op R(int v) { return {'R', v}; }
Op Q() { return {'Q', 0}; }

void report(const std::string& label, bool ok) {
    std::cout << label << ": " << (ok ? "OK" : "FAIL") << '\n';
}

// results: insert/remove -> 0 or 1, get_random -> returned value
void run_ops(const std::vector<Op>& ops, std::vector<int>& results, std::mt19937& rng) {
    RandomizedSet set;
    results.assign(ops.size(), 0);
    for (std::size_t i = 0; i < ops.size(); ++i) {
        switch (ops[i].kind) {
            case 'I': results[i] = set.insert(ops[i].value) ? 1 : 0; break;
            case 'R': results[i] = set.remove(ops[i].value) ? 1 : 0; break;
            default: results[i] = set.get_random(rng); break;
        }
    }
}

// Checks results against a model; returns the final model through `model`.
bool check_results(const std::vector<Op>& ops, const std::vector<int>& results,
                   std::unordered_set<int>& model) {
    model.clear();
    for (std::size_t i = 0; i < ops.size(); ++i) {
        switch (ops[i].kind) {
            case 'I':
                if (results[i] != (model.insert(ops[i].value).second ? 1 : 0)) return false;
                break;
            case 'R':
                if (results[i] != (model.erase(ops[i].value) > 0 ? 1 : 0)) return false;
                break;
            default:
                if (model.count(results[i]) == 0) return false;
                break;
        }
    }
    return true;
}

void test_script(const std::string& label, const std::vector<Op>& ops) {
    std::mt19937 rng(12345);
    std::vector<int> results;
    run_ops(ops, results, rng);
    std::unordered_set<int> model;
    report(label, check_results(ops, results, model));
}

// Same, and afterwards get_random must be roughly uniform over the final set.
void test_script_uniform(const std::string& label, const std::vector<Op>& ops) {
    std::mt19937 rng(12345);
    RandomizedSet set;
    std::unordered_set<int> model;
    bool ok = true;
    for (const Op& op : ops) {
        if (op.kind == 'I') {
            ok = ok && (set.insert(op.value) == model.insert(op.value).second);
        } else if (op.kind == 'R') {
            ok = ok && (set.remove(op.value) == (model.erase(op.value) > 0));
        } else {
            ok = ok && model.count(set.get_random(rng)) > 0;
        }
    }
    if (ok) {
        const int draws_per_element = 5000;
        const int total = draws_per_element * static_cast<int>(model.size());
        std::unordered_map<int, int> counts;
        for (int i = 0; i < total; ++i) {
            const int x = set.get_random(rng);
            if (model.count(x) == 0) { ok = false; break; }
            ++counts[x];
        }
        for (int v : model) {
            const int c = counts[v];
            if (c < draws_per_element * 8 / 10 || c > draws_per_element * 12 / 10) ok = false;
        }
    }
    report(label + " (uniform)", ok);
}

void test_big(const std::string& label, const std::vector<Op>& ops) {
    std::mt19937 rng(777);
    std::vector<int> results;
    const auto start = std::chrono::steady_clock::now();
    run_ops(ops, results, rng);
    const auto elapsed = std::chrono::steady_clock::now() - start;
    std::unordered_set<int> model;
    const bool correct = check_results(ops, results, model);
    const bool fast = elapsed < std::chrono::seconds(1);
    report(label + (fast ? "" : " (too slow)"), correct && fast);
}

	std::vector<int> storage_{};
	std::unordered_map<int, std::size_t> indexes_{};

public:
    bool insert(const int value) {
	    const auto it{indexes_.find(value)};
	    if (it != indexes_.end()) { return false; }

	    indexes_.emplace(value, storage_.size());
	    storage_.push_back(value);
	    
		return true;
	}

    bool remove(const int value) {
        const auto it{indexes_.find(value)};
        if (it == indexes_.end()) { return false; }

        storage_[rm_idx] = back_value;
        indexes_[back_value] = rm_idx;

        storage_.pop_back();
        indexes_.erase(value);

		return true;
