**Задача 6 из 6 (блок 5): медиана двух отсортированных массивов** (LeetCode 4, Hard)

Даны два массива `a` и `b`, каждый отсортирован по неубыванию. Нужно вернуть медиану объединённого массива. Если общий размер нечётный, это средний элемент. Если чётный, это среднее арифметическое двух средних. Массивы объединять в новый массив нельзя.

```cpp
double find_median(const std::vector<int>& a, const std::vector<int>& b);
```

**Ограничения**

- 0 ≤ m = `a.size()` ≤ 1 000 000, 0 ≤ n = `b.size()` ≤ 1 000 000, m + n ≥ 1.
- Значения лежат во всём диапазоне `int`, включая `INT_MIN` и `INT_MAX`.
- Один массив может быть пустым.
- Входные массивы менять нельзя.

**Вопросы интервьюера**

1. Какая сложность у решения «слить и взять середину»? Почему на тестах оно не пройдёт по времени?
2. Подойдёт ли решение двумя указателями за O(m + n) без дополнительной памяти? Что скажете про его время на 20 000 запросах?
3. Какой массив вы выбираете для бинарного поиска и зачем?
4. Что именно вы ищете бинарным поиском: значение или разбиение? Какое условие корректного разбиения, и как оно связано с монотонностью?
5. Как обрабатываете границы: пустую левую или правую часть, пустой массив целиком? Какие значения-«стражи» подставляете, и что с ними станет на `INT_MIN`/`INT_MAX`?
6. Где может переполниться `int`, когда считаете среднее двух средних?

**Целевая сложность:** O(log(min(m, n))) по времени, O(1) по памяти. Допустимо и O(log(m + n)) через поиск k-го элемента, но тогда будьте готовы объяснить, в чём разница.

**Ориентир по времени:** 40 минут. Задача самая сложная в блоке, так что не расстраивайтесь, если не уложитесь. Важнее, чтобы вы могли объяснить идею разбиения. Перед отправкой скомпилируйте с `-Wall -Wextra -Wconversion -Wsign-conversion` и один раз прогоните тесты.

namespace {
	double find_median([[maybe_unused]] const std::vector<int>& a,
					   [[maybe_unused]] const std::vector<int>& b) {
		return 0.0;
	}

	double expected_median(const std::vector<int>& a, const std::vector<int>& b) {
		std::vector<long long> merged(a.size() + b.size());
		std::merge(a.begin(), a.end(), b.begin(), b.end(), merged.begin());
		const std::size_t size{merged.size()};
		if (size % 2 == 1) return static_cast<double>(merged[size / 2]);
		// the sum of two ints fits in long long and in double without a loss
		return static_cast<double>(merged[size / 2 - 1] + merged[size / 2]) / 2.0;
	}

	void test_median(const std::string& label,
					 const std::vector<int>& a,
					 const std::vector<int>& b) {
		const auto expected{expected_median(a, b)};

		const auto result{find_median(a, b)};

		std::cout << std::format("{}: {}\n",
			label,
			result == expected ? "OK" : "FAIL") << std::flush;
	}

	// the same big pair is queried many times: an O(m + n) solution cannot finish in one second
	void test_median_many(const std::string& label,
						  const std::vector<int>& a,
						  const std::vector<int>& b,
						  int repeats) {
		const auto expected{expected_median(a, b)};

		const auto start{std::chrono::steady_clock::now()};
		bool ok{true};
		for (int i{0}; i < repeats; ++i) {
			// the compiler must not take the repeated call out of the loop
			asm volatile("" : : : "memory");
			if (find_median(a, b) != expected) {
				ok = false;
				break;
			}
			if (std::chrono::steady_clock::now() - start > std::chrono::seconds(1)) {
				ok = false;
				break;
			}
		}

		std::cout << std::format("{}: {}\n",
			label,
			ok ? "OK" : "FAIL") << std::flush;
	}

	// non-decreasing sequence: start, then each next value grows by a pseudo-random step in [0, max_step]
	std::vector<int> make_sorted(int size, int start, int max_step, unsigned seed) {
		std::vector<int> values;
		values.reserve(static_cast<std::size_t>(size));
		unsigned state{seed};
		int value{start};
		for (int i{0}; i < size; ++i) {
			values.push_back(value);
			state = state * 1103515245u + 12345u;
			value += static_cast<int>((state >> 8) % static_cast<unsigned>(max_step + 1));
		}
		return values;
	}
}

namespace {
	double find_median(const std::vector<int>& a,
					   const std::vector<int>& b) {
		
		if (a.size() > b.size()) {
			return find_median(b, a);
		}

		const int not_big_size{static_cast<int>(a.size())};
		const int big_size{static_cast<int>(b.size())};
        int lo{0};
        int hi{not_big_size};
        int total_half = (not_big_size + big_size + 1) / 2;
        
        while (lo < hi) {
	        const slash_a{lo + (hi - lo) / 2};
	        const slash_b{total_half - i};
            
            const auto max_left_a{
	            slash_a == 0
				? INT_MIN
				: a[static_cast<std::size_t>(slash_a - 1)]
            };
            const auto min_right_a{
	            slash_a == not_big_size
				? INT_MAX
				: a[static_cast<std::size_t>(slash_a)]
            }
            const auto max_left_b{
	            slash_b == 0
	            ? INT_MIN
	            : b[static_cast<std::size_t>(slash_b - 1)]
            };
            const auto min_right_b{
	            slash_b == big_size
	            ? INT_MAX
	            : b[static_cast<>(std::size_t)(shash_b)]
            };
            
            if (max_left_a <= min_right_b && max_left_b <= min_right_a) {
                if ((not_big_size + big_size) % 2 != 0) {
                    return std::max(max_left_a, max_left_b);
                }
                return (
	                std::max(max_left_a, max_left_b) +
	                std::min(min_right_a, min_right_b)
	            ) / 2.0;
            } else if (max_left_a > min_right_b) {
                hi = slash_a - 1;
            } else {
                lo = shash_a + 1;
            }
        }
		return 0.0;
	}
