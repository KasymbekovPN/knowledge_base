**A12: Sliding Window Maximum, `max_sliding_window` (пул А)**

Лимит 25 минут. Шаблон лежит в файле `MaxSlidingWindow_task.cpp`.

**Условие.** Дан массив `nums` и размер окна `k`. Окно из `k` подряд идущих элементов сдвигается слева направо на один элемент. Верните максимум в каждом положении окна: всего `n - k + 1` значений. Входной массив менять нельзя.

**Сигнатура:**
```cpp
std::vector<int> max_sliding_window(const std::vector<int>& nums, int k);
```

**Ограничения:** до 10^6 элементов; `1 <= k <= nums.size()`; значения любые `int`, включая крайние.

**Пример:** `nums = {1, 3, -1, -3, 5, 3, 6, 7}`, `k = 3` даёт `{3, 3, 5, 5, 6, 7}`.

**Вопросы интервьюера:**
1. Какая сложность у перебора максимума в каждом окне? Что можно улучшить с помощью `multiset`?
2. Как получить O(n) с монотонной очередью? Что в ней хранится: значения или индексы, и почему?
3. Когда нужно убирать элемент с переднего края очереди, а когда с заднего?
4. Что меняется, если в массиве есть равные значения?

**Цель:** O(n) по времени, O(k) дополнительной памяти.

Проверено: эталон проходит 22 из 22 и чист под санитайзерами; заглушка проходит 0; решение через `multiset` на O(n log k) проходит все тесты; решение с ошибкой на единицу при выбрасывании старого элемента проходит 11 из 22. Решение перебором за O(n·k) на больших тестах зависнет: прерви программу через `Ctrl+C`, это значит «too slow». Для компиляции подключи нужные заголовки сам. Пришли решение вложением.

```cpp
#include <iostream>
#include <cstddef>
#include <string>
#include <deque>
#include <vector>
#include <algorithm>
#include <chrono>

namespace {

// Return the maximum of every window of k consecutive elements of nums,
// from the leftmost window to the rightmost one (n - k + 1 values).
// 1 <= k <= nums.size().
std::vector<int> max_sliding_window(const std::vector<int>& nums, int k) {
	const std::size_t size{nums.size()};
	
	std::deque<int> indexes;
	std::vector<int> result;
	result.reserve(size - static_cast<std::size_t>(k) + 1);
	
	for (int i{0}; i < static_cast<int>(size); ++i) {
		while (!indexes.empty() && nums[indexes.back()] <= nums[i]) { indexes.pop_back(); }
		indexes.push_back(i);

		if (indexes.front() <= i - k) { indexes.pop_front(); }

		if (i >= k - 1) { result.push_back(nums[indexes.front()]); }
	}

    return result;
}

// Checker: naive O(n * k) for small tests, monotonic deque for big ones.
std::vector<int> naive(const std::vector<int>& nums, int k) {
    std::vector<int> result;
    const std::size_t width = static_cast<std::size_t>(k);
    for (std::size_t i = 0; i + width <= nums.size(); ++i) {
        int best = nums[i];
        for (std::size_t j = i; j < i + width; ++j) best = std::max(best, nums[j]);
        result.push_back(best);
    }
    return result;
}

std::vector<int> deque_reference(const std::vector<int>& nums, int k) {
    std::vector<int> result;
    std::deque<std::size_t> window;
    const std::size_t width = static_cast<std::size_t>(k);
    for (std::size_t i = 0; i < nums.size(); ++i) {
        while (!window.empty() && nums[window.back()] <= nums[i]) window.pop_back();
        window.push_back(i);
        if (window.front() + width <= i) window.pop_front();
        if (i + 1 >= width) result.push_back(nums[window.front()]);
    }
    return result;
}

void report(const std::string& label, bool ok) {
    std::cout << label << ": " << (ok ? "OK" : "FAIL") << '\n';
}

void test_window(const std::string& label, std::vector<int> nums, int k) {
    const std::vector<int> copy = nums;
    const bool ok = max_sliding_window(nums, k) == naive(nums, k) && nums == copy;
    report(label, ok);
}

void test_big(const std::string& label, const std::vector<int>& nums, int k) {
    const std::vector<int> expected = deque_reference(nums, k);
    const auto start = std::chrono::steady_clock::now();
    const std::vector<int> got = max_sliding_window(nums, k);
    const auto elapsed = std::chrono::steady_clock::now() - start;
    const bool fast = elapsed < std::chrono::seconds(1);
    report(label + (fast ? "" : " (too slow)"), got == expected && fast);
}

void test_all() {
    test_window("description example",
        {1, 3, -1, -3, 5, 3, 6, 7}, 3);
    test_window("one element",
        {5}, 1);
    test_window("k == 1",
        {4, -2, 7, 0}, 1);
    test_window("k == n",
        {4, -2, 7, 0}, 4);
    test_window("k == 2",
        {9, 8, 7, 6, 5}, 2);
    test_window("ascending",
        {1, 2, 3, 4, 5, 6}, 3);
    test_window("descending",
        {6, 5, 4, 3, 2, 1}, 3);
    test_window("all equal",
        {2, 2, 2, 2, 2}, 3);
    test_window("all negative",
        {-5, -3, -8, -1, -9, -2}, 2);
    test_window("maximum leaves the window",
        {10, 1, 2, 3, 4, 5}, 3);
    test_window("maximum at the window border",
        {1, 5, 1, 1, 5, 1, 1}, 3);
    test_window("equal maximums next to each other",
        {7, 7, 1, 7, 7, 1}, 2);
    test_window("extreme values",
        {-2147483647 - 1, 2147483647, -2147483647 - 1, 2147483647}, 2);
    test_window("two elements, k == 2",
        {3, 1}, 2);
    test_window("random 1: 10 elements from -5 to 5, k == 4",
        {3, -5, 0, 2, 5, -1, -3, 4, 1, -2}, 4);
    test_window("random 2: 15 elements from -9 to 9, k == 5",
        {-9, 4, 7, -2, 0, 9, 3, -6, 1, 8, -4, 5, -7, 2, 6}, 5);
    test_window("random 3: 30 elements from -3 to 3, k == 7",
        {1, -3, 0, 2, 2, -1, -2, 3, 0, 1, -3, -1, 2, 0, -2, 3, 1, -3, 0, 2, 3, -1, 1, 0, -2, 2, -3, 1, 0, 3}, 7);

    // ---- big tests ----
    {
        std::vector<int> nums(1000000);
        for (std::size_t i = 0; i < nums.size(); ++i) nums[i] = static_cast<int>(i);
        test_big("big: 1000000 ascending, k = 500000", nums, 500000);
    }
    {
        std::vector<int> nums(1000000);
        for (std::size_t i = 0; i < nums.size(); ++i) nums[i] = static_cast<int>(nums.size() - i);
        test_big("big: 1000000 descending, k = 500000", nums, 500000);
    }
    {
        std::vector<int> nums(1000000);
        unsigned x = 2024u;
        for (auto& v : nums) { x = x * 1103515245u + 12345u; v = static_cast<int>((x >> 8) % 2000001u) - 1000000; }
        test_big("big: 1000000 pseudo-random, k = 1000", nums, 1000);
    }
    {
        std::vector<int> nums(1000000);
        unsigned x = 99u;
        for (auto& v : nums) { x = x * 1103515245u + 12345u; v = static_cast<int>((x >> 8) % 2000001u) - 1000000; }
        test_big("big: 1000000 pseudo-random, k = 300000", nums, 300000);
    }
    {
        std::vector<int> nums(1000000, 7);
        test_big("big: 1000000 equal values, k = 1000000", nums, 1000000);
    }
}

}  // namespace

int main() {
    test_all();
    return 0;
}
```
