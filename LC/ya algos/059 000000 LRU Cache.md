## LRU Cache

Реализуйте кэш фиксированной ёмкости с вытеснением наименее недавно использованного ключа.

- `LruCache(capacity)` создаёт пустой кэш, `capacity ≥ 1`.
- `get(key)` возвращает значение ключа или `std::nullopt`, если такого ключа нет. Найденный ключ становится самым недавно использованным.
- `put(key, value)` сохраняет значение. Если ключ уже есть, значение обновляется, а ключ становится самым недавно использованным и ничего не вытесняется. Если ключа нет и кэш заполнен, сначала вытесняется ключ, использовавшийся давнее всех.

Обе операции должны работать за O(1) в среднем.

**Ограничения:** 1 ≤ capacity ≤ 10⁵, ключи и значения от 0 до 10⁹, до 10⁶ операций.

namespace {
	class LruCache {
	public:
		explicit LruCache([[maybe_unused]] std::size_t capacity) {
		}

		// the value of the key, or `std::nullopt` if there is no such key;
		// a found key becomes the most recently used one
		std::optional<int> get([[maybe_unused]] int key) {
			return std::nullopt;
		}

		// stores the value; if the key is new and the cache is full, the least recently used key is evicted first;
		// the key becomes the most recently used one
		void put([[maybe_unused]] int key, [[maybe_unused]] int value) {
		}
	};

	struct Operation {
		bool is_put;
		int key;
		int value;
	};

	Operation P(int key, int value) {
		return {true, key, value};
	}

	Operation G(int key) {
		return {false, key, 0};
	}

	// the results of all `get`s in order, -1 stands for a missing key
	std::vector<int> run(std::size_t capacity, const std::vector<Operation>& operations) {
		LruCache cache{capacity};
		std::vector<int> results;
		for (const Operation& operation : operations) {
			if (operation.is_put) {
				cache.put(operation.key, operation.value);
			} else {
				results.push_back(cache.get(operation.key).value_or(-1));
			}
		}
		return results;
	}

	void test_lru(const std::string& label,
				  std::size_t capacity,
				  const std::vector<Operation>& operations,
				  const std::vector<int>& expected) {
		std::cout << std::format("{}: {}\n",
			label,
			run(capacity, operations) == expected ? "OK" : "FAIL") << std::flush;
	}

	// and a map from that time to the key, so the oldest use is always the first element
	class Model {
	public:
		explicit Model(std::size_t capacity) : capacity_{capacity} {
		}

		int get(int key) {
			const auto found{entries_.find(key)};
			if (found == entries_.end()) {
				return -1;
			}
			touch(found);
			return found->second.value;
		}

		void put(int key, int value) {
			auto found{entries_.find(key)};
			if (found != entries_.end()) {
				found->second.value = value;
				touch(found);
				return;
			}
			if (entries_.size() == capacity_) {
				const auto oldest{by_time_.begin()};
				entries_.erase(oldest->second);
				by_time_.erase(oldest);
			}
			entries_[key] = Entry{value, ++clock_};
			by_time_[clock_] = key;
		}

	private:
		struct Entry {
			int value;
			long long time;
		};

		void touch(std::map<int, Entry>::iterator found) {
			by_time_.erase(found->second.time);
			found->second.time = ++clock_;
			by_time_[clock_] = found->first;
		}

		std::size_t capacity_;
		long long clock_{0};
		std::map<int, Entry> entries_;
		std::map<long long, int> by_time_;
	};

	std::vector<int> run_model(std::size_t capacity, const std::vector<Operation>& operations) {
		Model model{capacity};
		std::vector<int> results;
		for (const Operation& operation : operations) {
			if (operation.is_put) {
				model.put(operation.key, operation.value);
			} else {
				results.push_back(model.get(operation.key));
			}
		}
		return results;
	}

	void test_big(const std::string& label, std::size_t capacity, const std::vector<Operation>& operations) {
		const std::vector<int> expected{run_model(capacity, operations)};

		const auto start{std::chrono::steady_clock::now()};
		const std::vector<int> result{run(capacity, operations)};
		const auto elapsed{std::chrono::steady_clock::now() - start};

		const bool in_time{elapsed < std::chrono::seconds(1)};
		std::cout << std::format("{}: {}{}\n",
			label,
			result == expected ? "OK" : "FAIL",
			in_time ? "" : " (too slow)") << std::flush;
	}

	// `count` random operations over `keys` keys, `puts` percent of them are puts
	std::vector<Operation> random_operations(std::size_t count, int keys, int puts, unsigned seed) {
		std::mt19937 gen{seed};
		std::vector<Operation> operations;
		operations.reserve(count);
		for (std::size_t i{0}; i < count; ++i) {
			const int key{static_cast<int>(gen() % static_cast<unsigned>(keys))};
			if (static_cast<int>(gen() % 100) < puts) {
				operations.push_back(P(key, static_cast<int>(gen() % 1000000000)));
			} else {
				operations.push_back(G(key));
			}
		}
		return operations;
	}

	// the keys 0..cycle-1 over and over: get, then put
	std::vector<Operation> cyclic_operations(std::size_t count, int cycle) {
		std::vector<Operation> operations;
		operations.reserve(2 * count);
		for (std::size_t i{0}; i < count; ++i) {
			const int key{static_cast<int>(i % static_cast<std::size_t>(cycle))};
			operations.push_back(G(key));
			operations.push_back(P(key, static_cast<int>(i)));
		}
		return operations;
	}

		// a found key becomes the most recently used one

			const auto it{storage_.find(key)};

			}

	std::vector<Operation> cyclic_operations(std::size_t count, int cycle) {
		std::vector<Operation> operations;
		operations.reserve(2 * count);
		for (std::size_t i{0}; i < count; ++i) {
			const int key{static_cast<int>(i % static_cast<std::size_t>(cycle))};
			operations.push_back(G(key));
			operations.push_back(P(key, static_cast<int>(i)));
		}
		return operations;
	}
