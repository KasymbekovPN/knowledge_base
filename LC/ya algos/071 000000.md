**Условие.** Реализуйте префиксное дерево для слов из строчных латинских букв (длина ≥ 1).

- `insert(word)` добавляет слово. Повторная вставка того же слова ничего не меняет.
- `search(word)` возвращает `true`, если слово хранится целиком.
- `starts_with(prefix)` возвращает `true`, если есть хранимое слово, начинающееся с этого префикса.
- `count_with_prefix(prefix)` возвращает число **различных** хранимых слов, начинающихся с этого префикса. Слово, равное префиксу, тоже считается.

**Сигнатура:**

```cpp
class Trie {
public:
    void insert(const std::string& word);
    bool search(const std::string& word) const;
    bool starts_with(const std::string& prefix) const;
    int count_with_prefix(const std::string& prefix) const;
};
```

**Ограничения:** до 2·10^5 операций, слова и префиксы до 5·10^4 символов (тесты включают очень длинные слова).

**Пример:** `insert("apple")`, `search("apple")` даёт `true`, `search("app")` даёт `false`, `starts_with("app")` даёт `true`, `insert("app")`, `search("app")` даёт `true`, `count_with_prefix("app")` даёт 2, `count_with_prefix("a")` даёт 2, `count_with_prefix("b")` даёт 0.

**Цель:** `insert`, `search`, `starts_with`, `count_with_prefix` за O(длина слова); память O(суммарная длина слов · 26) в худшем случае для массива детей.

constexpr std::size_t ALPHABET_SIZE{26};

// Words and prefixes consist of lower-case latin letters and have length >= 1.
// insert adds a word (inserting the same word again changes nothing).
// search: is exactly this word stored?
// starts_with: is there a stored word that begins with this prefix?
// count_with_prefix: how many DIFFERENT stored words begin with this prefix
//   (a stored word is counted when it is equal to the prefix itself).
class Trie {
	struct Node{
		std::array<Node*, ALPHABET_SIZE> children{};
		bool is_end{false};
		int count{0};
	};
	
	Node* root_{new Node{}};
	
	std::size_t get_idx_(const char ch) const {
		return static_cast<std::size_t>(ch - 'a');
	}
	
	Node* find_node_(const std::string& word) const {
		Node* node{root_};
		for (const auto ch: word) {
			node = node->children[ger_idx_(ch)];
			if (node == nullptr) return nullptr;
		}
		
		return node;
	}

public:
	Trie() = default;
	Trie(const Trie&) = delete;
	Trie& operator=(const Trie&) = delete;
	
	~Trie() {
		std::vector<Node*> buffer{root_};
		while (!buffer.empty()) {
			Node* node{buffer.back()};
			buffer.pop_back();
			
			for (Node* child: node->children) {
				if (child != nullptr) buffer.push_back(child);
			}
			
			delete node;
		}
	}

    void insert(const std::string& word) {
	    if (search(word)) return;
    
	    Node* node{root_};
	    ++node->count;
	    for (const auto ch: word) {
		    Node*& child{node->children[get_idx_(ch)]};
		    if (child == nullptr) { child = new Node(); }
		    node = child;
		    ++node->count;
	    }
	    node->is_end = true;
    }

    bool search(const std::string& word) const {
	    const Node* node{find_node_(word)};
	    return node != nullptr && node->is_end;
	}

    bool starts_with(const std::string& word) const {
	    return find_node_(word) != nullptr;
	}

    int count_with_prefix(const std::string& word) const {
	    const Node* node{find_node_(word)};
	    return node != nullptr ? node->count : 0;
	}
};

struct Op {
    char kind;  // 'I' insert, 'S' search, 'P' starts_with, 'C' count_with_prefix
    std::string word;
};

Op I(const std::string& w) { return {'I', w}; }
Op S(const std::string& w) { return {'S', w}; }
Op P(const std::string& w) { return {'P', w}; }
Op C(const std::string& w) { return {'C', w}; }

void report(const std::string& label, bool ok) {
    std::cout << label << ": " << (ok ? "OK" : "FAIL") << '\n';
}

// Independent reference on a sorted set of words.
class Model {
public:
    void insert(const std::string& w) {
        if (words_.insert(w).second) sorted_.clear();
    }
    bool search(const std::string& w) const { return words_.count(w) > 0; }
    bool starts_with(const std::string& p) const { return count(p) > 0; }
    int count(const std::string& p) const {
        if (sorted_.empty()) sorted_.assign(words_.begin(), words_.end());
        // '{' follows 'z', so every word with prefix p lies in [p, p + '{')
        const auto lo = std::lower_bound(sorted_.begin(), sorted_.end(), p);
        const auto hi = std::lower_bound(sorted_.begin(), sorted_.end(), p + '{');
        return static_cast<int>(hi - lo);
    }

private:
    std::set<std::string> words_;
    mutable std::vector<std::string> sorted_;
};

// results: insert -> 0, search/starts_with -> 0 or 1, count -> the count
int apply(Trie& trie, const Op& op) {
    switch (op.kind) {
        case 'I': trie.insert(op.word); return 0;
        case 'S': return trie.search(op.word) ? 1 : 0;
        case 'P': return trie.starts_with(op.word) ? 1 : 0;
        default: return trie.count_with_prefix(op.word);
    }
}

int apply(Model& model, const Op& op) {
    switch (op.kind) {
        case 'I': model.insert(op.word); return 0;
        case 'S': return model.search(op.word) ? 1 : 0;
        case 'P': return model.starts_with(op.word) ? 1 : 0;
        default: return model.count(op.word);
    }
}

void test_script(const std::string& label, const std::vector<Op>& ops) {
    Trie trie;
    Model model;
    bool ok = true;
    for (const Op& op : ops) ok = (apply(trie, op) == apply(model, op)) && ok;
    report(label, ok);
}

void test_big(const std::string& label, const std::vector<Op>& ops) {
    std::vector<int> answers(ops.size());
    const auto start = std::chrono::steady_clock::now();
    {
        Trie trie;
        for (std::size_t i = 0; i < ops.size(); ++i) answers[i] = apply(trie, ops[i]);
    }
    const auto elapsed = std::chrono::steady_clock::now() - start;
    Model model;
    bool correct = true;
    for (std::size_t i = 0; i < ops.size() && correct; ++i) correct = answers[i] == apply(model, ops[i]);
    const bool fast = elapsed < std::chrono::seconds(1);
    report(label + (fast ? "" : " (too slow)"), correct && fast);
}
