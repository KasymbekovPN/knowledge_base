**Условие.** Дан массив целых `nums`. Верните все **уникальные** тройки значений `{a, b, c}`, у которых `a + b + c == 0`. Тройки возвращаются в любом порядке, внутри тройки порядок тоже любой. Одну и ту же позицию дважды использовать нельзя, а повторяющиеся по значениям тройки в ответе быть не должны.

**Сигнатура** (она уже стоит в шаблоне как заглушка):

```cpp

```

**Ограничения:** `3 <= nums.size() <= 3000`, `-1e5 <= nums[i] <= 1e5`. Массив можно менять.

namespace {
	using Triplets = std::vector<std::vector<int>>;

	Triplets three_sum([[maybe_unused]] std::vector<int>& nums) {
		return {};
	}

	Triplets normalize(Triplets triplets) {
		for (auto& triplet : triplets) {
			std::ranges::sort(triplet);
		}
		std::ranges::sort(triplets);
		return triplets;
	}

	Triplets expected_three_sum(std::vector<int> nums) {
		std::ranges::sort(nums);
		Triplets result;
		const std::size_t n{nums.size()};
		for (std::size_t i{0}; i + 2 < n; ++i) {
			if (i > 0 && nums[i] == nums[i - 1]) continue;
			std::size_t lo{i + 1};
			std::size_t hi{n - 1};
			while (lo < hi) {
				const long long sum{static_cast<long long>(nums[i]) + nums[lo] + nums[hi]};
				if (sum < 0) {
					++lo;
				} else if (sum > 0) {
					--hi;
				} else {
					result.push_back({nums[i], nums[lo], nums[hi]});
					++lo;
					--hi;
					while (lo < hi && nums[lo] == nums[lo - 1]) ++lo;
					while (lo < hi && nums[hi] == nums[hi + 1]) --hi;
				}
			}
		}
		return result;
	}

	void test_three_sum(const std::string& label,
						std::vector<int> nums,
						const Triplets& expected) {
		const auto result{normalize(three_sum(nums))};
		std::cout << std::format("{}: {}\n",
			label,
			result == normalize(expected) ? "OK" : "FAIL") << std::flush;
	}

	void test_big(const std::string& label,
				  std::vector<int> nums) {
		const auto expected{normalize(expected_three_sum(nums))};

		const auto start{std::chrono::steady_clock::now()};
		const auto result{normalize(three_sum(nums))};
		const auto elapsed{std::chrono::steady_clock::now() - start};

		const bool in_time{elapsed < std::chrono::seconds(1)};

namespace {
	using Triplets = std::vector<std::vector<int>>;

	Triplets three_sum(std::vector<int>& nums) {
		std::ranges::sort(nums);
		
		Triplets answer{};
		const auto size{static_cast<int>(nums.size())};
		for (int i{0}; i < size - 2; ++i) {
			if (i > 0 && nums[i] == nums[i - 1]) continue;
			
			int left{i + 1};
			int right{size - 1};
			while (left < right) {
				const auto sum{nums[i] + nums[left] + nums[right]};
				if (sum == 0) {
					std::vector<int> item{nums[i], nums[left], nums[right]};
					answer.push_back(std::move(item));
					
					++left;
					--right;
					
					while (left < right && nums[left] == nums[left - 1]) ++left;
					while (left < right && nums[right] == nums[right + 1]) --right;
				} else if (sum < 0) {
					++left;
				} else {
					--right;
				}
			}
		}
		return answer;
	}

	Triplets normalize(Triplets triplets) {
		for (auto& triplet : triplets) {
			std::ranges::sort(triplet);
		}
		std::ranges::sort(triplets);
		return triplets;
	}

	Triplets expected_three_sum(std::vector<int> nums) {
		std::ranges::sort(nums);
		Triplets result;
		const std::size_t n{nums.size()};
		for (std::size_t i{0}; i + 2 < n; ++i) {
			if (i > 0 && nums[i] == nums[i - 1]) continue;
			std::size_t lo{i + 1};
			std::size_t hi{n - 1};
			while (lo < hi) {
				const long long sum{static_cast<long long>(nums[i]) + nums[lo] + nums[hi]};
				if (sum < 0) {
					++lo;
				} else if (sum > 0) {
					--hi;
				} else {
					result.push_back({nums[i], nums[lo], nums[hi]});
					++lo;
					--hi;
					while (lo < hi && nums[lo] == nums[lo - 1]) ++lo;
					while (lo < hi && nums[hi] == nums[hi + 1]) --hi;
				}
			}
		}
		return result;
	}
