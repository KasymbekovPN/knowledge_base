## Базовая настройка проекта (делается один раз)

### vcpkg.json

```json
{
  "name": "boost-learning",
  "version": "0.1.0",
  "dependencies": [
    "boost-asio",
    "boost-smart-ptr",
    "boost-filesystem",
    "boost-regex",
    "boost-spirit",
    "boost-multiprecision",
    "boost-beast",
    "boost-graph",
    "boost-test",
    "boost-python",
    "boost-container",
    "boost-optional",
    "boost-variant",
    "boost-any"
  ]
}
```

### CMakeLists.txt

```cmake
cmake_minimum_required(VERSION 3.21)
project(boost_learning LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Подключение конкретных компонентов добавляется по мере изучения, например:
find_package(Boost REQUIRED COMPONENTS filesystem regex)

add_executable(app main.cpp)
target_link_libraries(app PRIVATE Boost::filesystem Boost::regex)
```

Конфигурация с toolchain-файлом vcpkg:

```bash
cmake -B build -S . -DCMAKE_TOOLCHAIN_FILE=[путь]/vcpkg/scripts/buildsystems/vcpkg.cmake
cmake --build build
```

Совет: положи `CMAKE_TOOLCHAIN_FILE` в `CMakePresets.json`, чтобы не писать длинную команду каждый раз.

### CMakePresets.json
```json
{  
  "version": 3,  
  "configurePresets": [  
    {  
      "name": "default",  
      "generator": "Ninja",  
      "binaryDir": "${sourceDir}/.build",  
      "toolchainFile": "$env{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake"  
    }  
  ]  
}
```
