### main.cpp
```cpp
// Versioning интерфейсов: как добавить новую возможность, не трогая  
// layout уже опубликованного интерфейса и не ломая уже собранных  
// потребителей ("старые бинарники" тут смоделированы как отдельный класс,  
// написанный ПРОТИВ старого интерфейса и не подозревающий о новом).  
  
#include <iostream>  
#include <format>  
  
// ---------------------------------------------------------------------------  
// V1 - опубликованный интерфейс. У него уже есть внешние реализации  
// (представим - в закрытых бинарниках, которые мы не можем пересобрать).  
// Его vtable-layout НИКОГДА больше не меняется - ни один метод не  
// добавляется, не удаляется, не переставляется.  
// ---------------------------------------------------------------------------  
class IRenderV1 {  
public:  
    virtual ~IRenderV1() = default;  
    virtual void drawPixel(int x, int y) = 0;  
};  
  
// ---------------------------------------------------------------------------  
// V2 - НОВЫЙ интерфейс, расширяющий V1 через наследование, а не изменяющий  
// его. Существующие реализации V1 эту иерархию вообще не видят - для них  
// ничего не поменялось ни на уровне исходников, ни на уровне ABI.  
// ---------------------------------------------------------------------------  
class IRenderV2: public IRenderV1 {  
public:  
    virtual void drawLine(int x1, int y1, int x2, int y2) = 0;  
};  
  
// ---------------------------------------------------------------------------  
// "Старый" плагин - собран когда-то давно, знает только про V1.  
// Пересобирать/трогать его исходники мы не можем и не должны.  
// ---------------------------------------------------------------------------  
class OldSoftwareRenderer: public IRenderV1 {  
public:  
    void drawPixel(int x, int y) override {  
        std::cout << std::format("  [software] pixel({}, {})\n", x, y);  
    }
};  
  
// ---------------------------------------------------------------------------  
// "Новый" плагин - собран сегодня, реализует расширенный V2.  
// ---------------------------------------------------------------------------  
class NewGpuRenderer: public IRenderV2 {  
public:  
    void drawPixel(int x, int y) override {  
        std::cout << std::format("  [gpu] pixel({}, {})\n", x, y);  
    }    void drawLine(int x1, int y1, int x2, int y2) override {  
        std::cout << std::format("  [gpu] line ({}, {}) -> ({}, {})", x1, y1, x2, y2);  
    }
};  
  
// ---------------------------------------------------------------------------  
// Host работает с базовым IRendererV1& - гарантированно совместим с ЛЮБОЙ  
// реализацией, старой или новой. Дополнительные возможности V2  
// запрашиваются через dynamic_cast как необязательный "capability query" -  
// если объект реально реализует V2, используем более эффективный путь;  
// если нет (старый плагин) - работаем через гарантированный минимум V1.  
// ---------------------------------------------------------------------------  
void render(IRenderV1& renderer) {  
    renderer.drawPixel(1, 1);  
  
    if (auto* it = dynamic_cast<IRenderV2*>(&renderer)) {  
        it->drawLine(0, 0, 10, 10);  
    } else {  
        std::cout << "  drawLine is not support\n";  
    }
}  
  
int main() {  
    OldSoftwareRenderer oldRenderer;  
    NewGpuRenderer newRenderer;  
  
    render(oldRenderer);  
    render(newRenderer);  
  
    return 0;  
}
```

**Почему это вообще сложно**

Пока интерфейс используется только внутри одного проекта, который целиком пересобирается при каждом изменении — версионирование не проблема: поменял метод, пересобрал всё, готово. Проблема начинается, когда есть внешние потребители — отдельно скомпилированные бинарники (плагины, как в нашем `plugin_demo`, сторонние библиотеки, закрытый код партнёров), которые нельзя заставить пересобраться одновременно с тобой. Тогда любое изменение интерфейса нужно оценивать не "компилируется ли", а "продолжит ли работать код, скомпилированный ДО этого изменения, без пересборки".

**Что реально ломает бинарную совместимость (ABI)**

Удаление или перестановка виртуальных методов — всегда катастрофа: у всех остальных методов после места правки сдвигается смещение в vtable, старый бинарник будет вызывать не тот метод. Добавление нового виртуального метода **в середину** — то же самое. Добавление **в конец** — относительно безопасно для layout существующих методов (их смещения не меняются), но если это `= 0` (чисто виртуальный) — это ломает **исходники** всех, кто наследует интерфейс: их класс больше не полон, не компилируется, пока не добавят реализацию. Если добавить с телом по умолчанию (не чисто виртуальный) — старые реализации остаются компилируемыми и рабочими, они просто не переопределяют новый метод и получают поведение по умолчанию.

**Стратегия из демо: расширение через отдельный производный интерфейс**

Вместо того чтобы трогать `IRendererV1` вообще, `IRendererV2` — новый, отдельный тип, наследующий `V1` и добавляющий `drawLine()`. `IRendererV1`, у которого уже есть внешние реализации (`OldSoftwareRenderer` — как будто собран когда-то давно и недоступен для пересборки), никогда не меняется ни на байт — ни его vtable-layout, ни его исходники. `render()` работает с базовым `IRendererV1&` — гарантированный минимум, который есть у абсолютно любой реализации, старой или новой, — а `dynamic_cast<IRendererV2*>` работает как необязательный "capability query": "а вдруг этот конкретный объект умеет больше". Тест это подтвердил буквально: `OldSoftwareRenderer` прошла через безопасный fallback (эмуляция линии через отдельные пиксели), `NewGpuRenderer` — через настоящий `drawLine()`, и оба варианта работают через один и тот же вызывающий код, без единого `if`, завязанного на конкретный тип.

**Альтернатива — явный номер версии (как в `plugin_demo`)**

В примере с `dlopen`-плагинами мы использовали не `dynamic_cast`, а явную функцию `pluginApiVersion()`, сверяемую до вызова чего-либо ещё. У этого подхода есть причина: `dynamic_cast` полагается на RTTI — сравнение `typeid`, которое в общем случае корректно работает между разными `.so`/`.dll` в рамках одного процесса на современных платформах, но исторически имело нюансы с видимостью символов между модулями (особенно при `RTLD_LOCAL` или в кодовых базах, где RTTI вообще отключён флагом `-fno-rtti` — частая практика в игровых движках ради размера бинарника и производительности). Явный числовой version-check не требует RTTI вообще и однозначно понятен на любой платформе — поэтому для C-ABI границы (`dlopen`/`LoadLibrary`, как в нашем плагинном примере) он надёжнее; `dynamic_cast`-based capability query удобнее внутри одной программы на чистом C++, где RTTI включён по умолчанию и это не C-style граница.

**Что делать с данными, а не только с методами**

Если через границу передаётся не интерфейс, а конкретный класс по значению (не через указатель на абстракцию) — это вопрос уже не про виртуальные методы, а про `sizeof`/layout, и решается тем, что разбирали в теме pImpl: публичный класс хранит только `unique_ptr<Impl>`, и `Impl` можно свободно расширять новыми полями сколько угодно раз, не трогая ABI внешнего класса вообще — это ортогональная, но не менее важная ось версионирования: методы версионируются через отдельные интерфейсы (`V1`/`V2`), данные — через pImpl.

**Итоговое практическое правило**

Для долгоживущего опубликованного интерфейса: никогда не удаляй и не переставляй существующие виртуальные методы; новую функциональность добавляй либо через отдельный производный интерфейс (`V2 : V1`), либо, если действительно нужно расширить сам `V1`, — только методом с реализацией по умолчанию в конце объявления, никогда не чисто виртуальным; для C-ABI границ (плагины через `dlopen`/`LoadLibrary`) — явный числовой version-check перед любым вызовом; для передаваемых по значению классов — pImpl, чтобы рост данных не трогал ABI.
